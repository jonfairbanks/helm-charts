1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "rick.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  echo "The load balancer address may take a few minutes to become available."
  kubectl get --namespace {{ .Release.Namespace }} service {{ include "rick.fullname" . }} --watch
{{- else }}
  echo "Visit http://127.0.0.1:8080"
  kubectl --namespace {{ .Release.Namespace }} port-forward service/{{ include "rick.fullname" . }} 8080:{{ .Values.service.port }}
{{- end }}
